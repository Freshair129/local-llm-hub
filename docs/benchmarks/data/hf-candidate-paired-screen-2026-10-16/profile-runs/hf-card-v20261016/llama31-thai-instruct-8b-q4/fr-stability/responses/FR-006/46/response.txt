// trace:implements FR-006
#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    let columns: Vec<&str> = line.split(',').collect();
    if columns.len() < 5 {
        return Err(format!("Invalid CSV line: {}", line));
    }

    let name = columns[0].to_string();
    let vram_used_mb = columns[1].parse().map_err(|_| format!("Invalid VRAM Used: {}", columns[1]))?;
    let vram_total_mb = columns[2].parse().map_err(|_| format!("Invalid VRAM Total: {}", columns[2]))?;
    let temp_c = columns[3].parse().map_err(|_| format!("Invalid Temperature: {}", columns[3]))?;
    let util_pct = columns[4].parse().map_err(|_| format!("Invalid Utilization: {}", columns[4]))?;

    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
