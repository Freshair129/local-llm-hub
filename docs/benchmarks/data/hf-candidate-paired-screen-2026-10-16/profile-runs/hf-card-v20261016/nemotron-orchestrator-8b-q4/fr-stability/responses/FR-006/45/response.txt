#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    // Trim leading/trailing whitespace
    let line = line.trim();
    
    // Check if line is empty
    if line.is_empty() {
        return Err("Empty input line".to_string());
    }
    
    // Split by commas
    let parts: Vec<&str> = line.split(',').collect();
    
    // Check we have exactly 5 columns
    if parts.len() != 5 {
        return Err(format!("Expected 5 columns, found {}", parts.len()));
    }
    
    // Parse each column
    let name = parts[0].trim().to_string();
    
    let vram_used_mb = match parts[1].trim().parse::<u64>() {
        Ok(v) => v,
        Err(_) => return Err("Failed to parse VRAM used".to_string()),
    };
    
    let vram_total_mb = match parts[2].trim().parse::<u64>() {
        Ok(v) => v,
        Err(_) => return Err("Failed to parse VRAM total".to_string()),
    };
    
    let temp_c = match parts[3].trim().parse::<u32>() {
        Ok(v) => v,
        Err(_) => return Err("Failed to parse temperature".to_string()),
    };
    
    let util_pct = match parts[4].trim().parse::<u32>() {
        Ok(v) => v,
        Err(_) => return Err("Failed to parse utilization".to_string()),
    };
    
    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
