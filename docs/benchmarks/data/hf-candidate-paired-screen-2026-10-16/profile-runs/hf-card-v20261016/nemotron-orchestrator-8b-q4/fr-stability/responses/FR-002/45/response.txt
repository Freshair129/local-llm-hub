pub fn normalize_model_name(raw: &str) -> String {
    let mut result = raw.to_string();
    
    // Strip registry prefixes
    if result.starts_with("registry.ollama.ai/library/") {
        result = result["registry.ollama.ai/library/".len()..].to_string();
    } else if result.starts_with("library/") {
        result = result["library/".len()..].to_string();
    } else if result.starts_with("hf.co/") {
        result = result["hf.co/".len()..].to_string();
    }
    
    // Strip quantization suffixes
    let quantization_suffixes = ["-q4_0", "-q4_k_m", "-q4_k_s", ":q4_0", ":q4_k_m", ":q4_k_s"];
    
    for &suf in &quantization_suffixes {
        let lower_result = result.to_lowercase();
        if lower_result.ends_with(suf) {
            result = result[..result.len() - suf.len()].to_string();
            break; // Only remove the first matching suffix
        }
    }
    
    result.to_lowercase()
}
