    pub fn normalize_model_name(raw: &str) -> String {
        let mut name = raw.to_string();

        // 1. Strip registry prefixes
        let prefixes = [
            "registry.ollama.ai/library/",
            "library/",
            "hf.co/",
        ];

            if name.starts_with(prefix) {

                break;

                name = lower_name[..lower_name.len() - suffix.len()].to_string();
                break;
            }
