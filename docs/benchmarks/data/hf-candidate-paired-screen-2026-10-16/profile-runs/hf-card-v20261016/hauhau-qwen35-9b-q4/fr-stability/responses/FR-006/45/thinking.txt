        let util_pct = parts[4].trim().parse::<u32>().map_err(|e| format!("util_pct parse error: {}", e))?;

        Ok(GpuStats {
            name,
            vram_used_mb,
            vram_total_mb,
            temp_c,
            util_pct,
        })
    }
