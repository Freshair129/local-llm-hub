    pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
        let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();

        if parts.len() < 5 {
            return Err(format!("Expected at least 5 columns, found {}", parts.len()));
        }

        let name = parts[0].to_string();
        let vram_used_mb = parts[1].parse::<u64>().map_err(|e| e.to_string())?;
        let vram_total_mb = parts[2].parse::<u64>().map_err(|e| e.to_string())?;
        let temp_c = parts[3].parse::<u32>().map_err(|e| e.to_string())?;
        let util_pct = parts[4].parse::<u32>().map_err(|e| e.to_string())?;

        Ok(GpuStats {
            name,
            vram_used_mb,
            vram_total_mb,
            temp_c,
            util_pct,
        })
    }
