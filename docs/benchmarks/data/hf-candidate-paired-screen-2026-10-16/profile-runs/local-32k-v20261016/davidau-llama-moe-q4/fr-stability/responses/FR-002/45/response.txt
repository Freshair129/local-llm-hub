/// trace:implements FR-002
pub fn normalize_model_name(raw: &str) -> String {
    // Step 1: Strip registry prefixes
    let raw = raw.replace("registry.ollama.ai/library/", "").replace("library/", "");

    // Step 2: Strip quantization suffixes
    let raw = raw.to_lowercase().replace(-"q4_0", "").replace(-"q4_k_m", "").replace(-"q4_k_s", "").replace(":q4_0", "").replace(":q4_k_m", "").replace(":q4_k_s", "");

    // Step 3: Convert result to lowercase
    raw.to_lowercase()
}
