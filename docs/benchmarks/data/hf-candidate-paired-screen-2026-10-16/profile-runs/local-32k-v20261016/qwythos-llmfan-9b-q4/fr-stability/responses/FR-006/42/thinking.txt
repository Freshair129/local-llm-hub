```rust
// trace:implements FR-006
pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    let parts: Vec<&str> = line.split(',').collect();
    if parts.len() != 5 {
        return Err(format!("Invalid CSV format: expected 5 columns, got {}", parts.len()));
    }

    let name = parts[0].trim().to_string();
    let vram_used_mb = parts[1].parse::<u64>().map_err(|_| "Failed to parse vram_used_mb")?;
    let vram_total_mb = parts[2].parse::<u64>().map_err(|_| "Failed to parse vram_total_mb")?;
    let temp_c = parts[3].parse::<u32>().map_err(|_| "Failed to parse temp_c")?;
    let util_pct = parts[4].parse::<u32>().map_err(|_| "Failed to parse util_pct")?;

    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
```
