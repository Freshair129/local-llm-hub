#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    // Remove quotes and trim whitespace
    let cleaned_line = line.replace("\"", "").trim();
    
    // Split by commas
    let parts: Vec<&str> = cleaned_line.split(',').collect();
    
    // Check for exactly 5 columns
    if parts.len() != 5 {
        return Err(format!("CSV line must have exactly 5 columns, found {}", parts.len()));
    }
    
    let name = parts[0].trim().to_string();
    
    // Parse VRAM used
    let vram_used_mb = match parts[1].trim().parse::<u64>() {
        Ok(val) if val <= u64::MAX => val,
        _ => return Err("Failed to parse VRAM used (must be u64)".to_string()),
    };
    
    // Parse VRAM total
    let vram_total_mb = match parts[2].trim().parse::<u64>() {
        Ok(val) if val <= u64::MAX => val,
        _ => return Err("Failed to parse VRAM total (must be u64)".to_string()),
    };
    
    // Parse temperature
    let temp_c = match parts[3].trim().parse::<u32>() {
        Ok(val) if val <= u32::MAX => val,
        _ => return Err("Failed to parse temperature (must be u32)".to_string()),
    };
    
    // Parse utilization
    let util_pct = match parts[4].trim().parse::<u32>() {
        Ok(val) if val <= 100 => val,
        _ => return Err("Utilization must be between 0 and 100".to_string()),
    };
    
    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
