#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

// trace:implements FR-006
pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    let tokens: Vec<String> = line
        .split(',')
        .map(|s| s.trim().to_string())
        .collect();

    if tokens.len() < 5 {
        return Err(format!("Expected 5 columns, got {}", tokens.len()));
    }

    let name = tokens[0].clone();
    let vram_used_mb = tokens[1].parse::<u64>().map_err(|e| e.to_string())?;
    let vram_total_mb = tokens[2].parse::<u64>().map_err(|e| e.to_string())?;
    let temp_c = tokens[3].parse::<u32>().map_err(|e| e.to_string())?;
    let util_pct = tokens[4].parse::<u32>().map_err(|e| e.to_string())?;

    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
