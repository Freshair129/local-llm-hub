   ```rust
   // trace:implements FR-006
   pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
       // split by comma, trim each field
       let fields: Vec<&str> = line.split(',').map(|s| s.trim()).collect();

       // ensure at least 5 columns
       if fields.len() < 5 {
           return Err(format!("Expected 5 columns, got {}", fields.len()));
       }

       // parse each field, mapping ParseIntError to a descriptive Err
       let name = fields[0];
       let vram_used = fields[1].parse::<u64>()
           .map_err(|_| format!("Failed to parse VRAM used: {}", fields[1]));
       let vram_total = fields[2].parse::<u64>()
           .map_err(|_| format!("Failed to parse VRAM total: {}", fields[2]));
       let temp = fields[3].parse::<u32>()
           .map_err(|_| format!("Failed to parse temperature: {}", fields[3]));
       let util = fields[4].parse::<u32>()
           .map_err(|_| format!("Failed to parse utilization: {}", fields[4]));

       // combine into a Result, chaining the parsed values
       Ok(GpuStats {
           name: name.to_string(),
           vram_used_mb: vram_used?,
           vram_total_mb: vram_total?,
           temp_c: temp?,
           util_pct: util?,
       })
   }
   ```

   - `name` is a plain `&str`, so we `.to_string()` to get a `String`.
   - For the numeric fields, we map `ParseIntError` to a clear message, then `?` to propagate if any fail.
