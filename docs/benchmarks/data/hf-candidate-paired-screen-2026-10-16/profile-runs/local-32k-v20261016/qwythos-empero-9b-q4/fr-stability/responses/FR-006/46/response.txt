#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

// trace:implements FR-006
pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();
    if parts.len() < 5 {
        return Err(format!("Expected 5 columns, got {}", parts.len()));
    }

    let name = parts[0].to_string();
    let vram_used_mb = parts[1].parse::<u64>()
        .map_err(|_| format!("Failed to parse column 1 as u64: '{}'", parts[1]))?;
    let vram_total_mb = parts[2].parse::<u64>()
        .map_err(|_| format!("Failed to parse column 2 as u64: '{}'", parts[2]))?;
    let temp_c = parts[3].parse::<u32>()
        .map_err(|_| format!("Failed to parse column 3 as u32: '{}'", parts[3]))?;
    let util_pct = parts[4].parse::<u32>()
        .map_err(|_| format!("Failed to parse column 4 as u32: '{}'", parts[4]))?;

    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
