pub fn normalize_model_name(raw: &str) -> String {
    // trace:implements FR-002

    let mut name = raw;

    // Strip registry prefixes (case-insensitive matching)
    if let Some(pos) = name.to_lowercase().find("registry.ollama.ai/library/") {
        name = &name[pos + "registry.ollama.ai/library/".len()..];
    } else if let Some(pos) = name.to_lowercase().find("library/") {
        name = &name[pos + "library/".len()..];
    } else if let Some(pos) = name.to_lowercase().find("hf.co/") {
        name = &name[pos + "hf.co/".len()..];
    }

    // Strip quantization suffixes (case-insensitive)
    let quant_suffixes = [
        "-q4_0", "-q4_k_m", "-q4_k_s",
        ":q4_0", ":q4_k_m", ":q4_k_s",
    ];

    for suffix in quant_suffixes.iter() {
        let suffix_lower = suffix.to_lowercase();
        if let Some(pos) = name.to_lowercase().find(&suffix_lower) {
            name = &name[pos + suffix.len()..];
            break;
        }
    }

    // Convert to lowercase
    name.to_lowercase()
}
