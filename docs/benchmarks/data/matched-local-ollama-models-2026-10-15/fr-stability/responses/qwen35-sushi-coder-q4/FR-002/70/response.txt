// trace:implements FR-002
pub fn normalize_model_name(raw: &str) -> String {
    let mut normalized = raw.to_string();
    
    // Strip registry prefixes (case-insensitive)
    let prefixes = ["registry.ollama.ai/library/", "library/", "hf.co/"];
    for prefix in &prefixes {
        if normalized.to_lowercase().starts_with(prefix) {
            normalized = normalized[prefix.len()..].to_string();
            break;
        }
    }
    
    // Strip quantization suffixes (case-insensitive)
    let suffixes = ["-q4_0", "-q4_k_m", "-q4_k_s", ":q4_0", ":q4_k_m", ":q4_k_s"];
    for suffix in &suffixes {
        let suffix_lower = suffix.to_lowercase();
        if normalized.to_lowercase().ends_with(&suffix_lower) {
            normalized = normalized[..normalized.len() - suffix.len()].to_string();
            break;
        }
    }
    
    // Convert to lowercase
    normalized = normalized.to_lowercase();
    
    normalized
}
