```rust
#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

// trace:implements FR-006
pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    // 1. Check for empty line
    if line.is_empty() {
        return Err("Input line is empty".to_string());
    }

    // 2. Split by comma
    let parts: Vec<&str> = line.split(',').collect();

    // 3. Validate column count
    if parts.len() != 5 {
        return Err(format!("Expected 5 columns, found {}", parts.len()));
    }

    // 4. Parse fields into GpuStats
    // We return Err immediately if parsing fails, avoiding unwrap/panic.
    let name = parts[0].to_string();
    let vram_used = parts[1].parse::<u64>()
        .map_err(|_| format!("Failed to parse vram_used_mb: {}", parts[1]))?;
    
    let vram_total = parts[2].parse::<u64>()
        .map_err(|_| format!("Failed to parse vram_total_mb: {}", parts[2]))?;
        
    let temp = parts[3].parse::<u32>()
        .map_err(|_| format!("Failed to parse temp_c: {}", parts[3]))?;
        
    let util = parts[4].parse::<u32>()
        .map_err(|_| format!("Failed to parse util_pct: {}", parts[4]))?;

    Ok(GpuStats {
        name,
        vram_used_mb: vram_used,
        vram_total_mb: vram_total,
        temp_c: temp,
        util_pct: util,
    })
}
```
