// trace:implements FR-006

#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    // Split by comma
    let parts: Vec<&str> = line.split(',').collect();
    
    // Guard rail: Check column count
    if parts.len() < 5 {
        return Err(format!("Expected 5 columns, got {}", parts.len()));
    }
    
    // Parse name (String)
    let name = parts[0].trim().to_string();
    
    // Parse vram_used_mb (u64)
    let vram_used_mb = parts[1].trim().parse::<u64>()
        .map_err(|_| format!("Failed to parse vram_used_mb from '{}'", parts[1]))?;
    
    // Parse vram_total_mb (u64)
    let vram_total_mb = parts[2].trim().parse::<u64>()
        .map_err(|_| format!("Failed to parse vram_total_mb from '{}'", parts[2]))?;
    
    // Parse temp_c (u32)
    let temp_c = parts[3].trim().parse::<u32>()
        .map_err(|_| format!("Failed to parse temp_c from '{}'", parts[3]))?;
    
    // Parse util_pct (u32)
    let util_pct = parts[4].trim().parse::<u32>()
        .map_err(|_| format!("Failed to parse util_pct from '{}'", parts[4]))?;
    
    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}

// Example usage:
fn main() {
    let line = "NVIDIA GeForce RTX 3060, 5800, 12288, 35, 25";
    match parse_gpu_csv(line) {
        Ok(stats) => println!("{:?}", stats),
        Err(e) => println!("Error: {}", e),
    }
}
