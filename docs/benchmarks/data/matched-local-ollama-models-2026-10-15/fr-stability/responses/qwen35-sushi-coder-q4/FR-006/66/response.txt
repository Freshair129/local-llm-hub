// trace:implements FR-006

#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    // Split CSV line by comma
    let fields: Vec<&str> = line.split(',').collect();
    
    // Guard rail: validate field count
    if fields.len() < 5 {
        return Err(format!("Expected 5 fields, got {}", fields.len()));
    }
    
    let name = fields[0].trim().to_string();
    
    // Parse numeric fields with proper error handling
    let vram_used_mb = fields[1]
        .trim()
        .parse::<u64>()
        .map_err(|e| format!("Failed to parse vram_used_mb: {}", e))?;
    
    let vram_total_mb = fields[2]
        .trim()
        .parse::<u64>()
        .map_err(|e| format!("Failed to parse vram_total_mb: {}", e))?;
    
    let temp_c = fields[3]
        .trim()
        .parse::<u32>()
        .map_err(|e| format!("Failed to parse temp_c: {}", e))?;
    
    let util_pct = fields[4]
        .trim()
        .parse::<u32>()
        .map_err(|e| format!("Failed to parse util_pct: {}", e))?;
    
    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}

// Example usage and test:
fn main() {
    let line = "NVIDIA GeForce RTX 3060, 5800, 12288, 35, 25";
    match parse_gpu_csv(line) {
        Ok(stats) => println!("{:?}", stats),
        Err(e) => println!("Error: {}", e),
    }
}
