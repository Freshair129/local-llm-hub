#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    let fields: Vec<&str> = line.trim().split(',').collect();

    if fields.len() < 5 {
        return Err(String::from("CSV line must contain at least 5 columns"));
    }

    let name = fields[0].trim().to_string();

    let vram_used_mb = match fields[1].trim().parse::<u64>() {
        Ok(val) => val,
        Err(_) => return Err(String::from("Failed to parse vram_used_mb as u64")),
    };

    let vram_total_mb = match fields[2].trim().parse::<u64>() {
        Ok(val) => val,
        Err(_) => return Err(String::from("Failed to parse vram_total_mb as u64")),
    };

    let temp_c = match fields[3].trim().parse::<u32>() {
        Ok(val) => val,
        Err(_) => return Err(String::from("Failed to parse temp_c as u32")),
    };

    let util_pct = match fields[4].trim().parse::<u32>() {
        Ok(val) => val,
        Err(_) => return Err(String::from("Failed to parse util_pct as u32")),
    };

    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
