// trace:implements FR-006
#[derive(Debug, PartialEq)]
pub struct GpuStats {
    pub name: String,
    pub vram_used_mb: u64,
    pub vram_total_mb: u64,
    pub temp_c: u32,
    pub util_pct: u32,
}

// trace:implements FR-006
pub fn parse_gpu_csv(line: &str) -> Result<GpuStats, String> {
    let mut fields = line.split(',');

    let name = fields
        .next()
        .ok_or("Missing GPU name".to_string())?
        .trim()
        .to_string();

    let vram_used_mb_str = fields
        .next()
        .ok_or("Missing VRAM used value".to_string())?
        .trim();

    let vram_total_mb_str = fields
        .next()
        .ok_or("Missing VRAM total value".to_string())?
        .trim();

    let temp_c_str = fields
        .next()
        .ok_or("Missing temperature value".to_string())?
        .trim();

    let util_pct_str = fields
        .next()
        .ok_or("Missing utilization value".to_string())?
        .trim();

    // Check for additional unexpected fields
    if fields.next().is_some() {
        return Err("Too many columns".to_string());
    }

    let vram_used_mb = vram_used_mb_str
        .parse::<u64>()
        .map_err(|_| "Failed to parse VRAM used as u64".to_string())?;

    let vram_total_mb = vram_total_mb_str
        .parse::<u64>()
        .map_err(|_| "Failed to parse VRAM total as u64".to_string())?;

    let temp_c = temp_c_str
        .parse::<u32>()
        .map_err(|_| "Failed to parse temperature as u32".to_string())?;

    let util_pct = util_pct_str
        .parse::<u32>()
        .map_err(|_| "Failed to parse utilization as u32".to_string())?;

    Ok(GpuStats {
        name,
        vram_used_mb,
        vram_total_mb,
        temp_c,
        util_pct,
    })
}
